function saveFreightRates(){
  const port=PORTS[$('freightPort').selectedIndex];
  if(!port) return;
  const rates={
    sea:Number($('freightSea').value),
    rail:Number($('freightRail').value),
    air:Number($('freightAir').value)
  };
  if(Object.values(rates).some(rate=>!Number.isFinite(rate)||rate<0)||rates.sea>rates.rail||rates.rail>rates.air){
    $('freightRateError').textContent=I18N[LANG].freightRateError;

  FREIGHT_RATES[port.en]=rates;
  localStorage.setItem(FREIGHT_RATE_KEY,JSON.stringify(FREIGHT_RATES));
  $('freightRateError').textContent='';
  freightCalc();
  update();
}

function num(id,def){ const v=parseFloat($(id).value); return isNaN(v)?def:v; }
function calc(){
  const purchase=num('purchase',118.57), vat=num('vatRate',13)/100, reb=num('rebateRate',13)/100;
  const domestic=num('domestic',2.57), packCert=num('packCert',0.71), margin=num('margin',15)/100;
  const freight=freightCalc(), insRate=num('insRate',0.3)/100, addOn=1+num('insAddOn',10)/100;
  const qty=Math.max(1,Math.floor(num('qty',300))), samplePrice=num('samplePrice',280);
  const rebate=purchase/(1+vat)*reb;
  const actual=purchase-rebate;
  const totalCost=actual+domestic+packCert;
  const exw=(actual+packCert)*(1+margin);
  const fob=totalCost*(1+margin);
  const cif=(fob+freight)/(1-addOn*insRate);
  const ins=cif*addOn*insRate;
  const isSample=qty<50;
  const quoteExw=isSample?samplePrice-(fob-exw):exw;
  const quoteFob=isSample?samplePrice:fob;
  const quoteCif=(quoteFob+freight)/(1-addOn*insRate);
  const quoteIns=quoteCif*addOn*insRate;
  const quoteFreight=quoteFob+freight;
  const basis=INCOTERMS[TERM].basis;
  const unitPrice=basis==='exw'?quoteExw:basis==='fob'?quoteFob:basis==='freight'?quoteFreight:basis==='insured'?quoteCif:quoteFreight;
  return {
    purchase,rebate,actual,totalCost,domestic,packCert,profit:totalCost*margin,
    exw:quoteExw,fob:quoteFob,cif:quoteCif,ins:quoteIns,qty,freight,isSample,samplePrice,unitPrice
  };
}

function numToWords(num){
  const ones=['ZERO','ONE','TWO','THREE','FOUR','FIVE','SIX','SEVEN','EIGHT','NINE','TEN','ELEVEN','TWELVE','THIRTEEN','FOURTEEN','FIFTEEN','SIXTEEN','SEVENTEEN','EIGHTEEN','NINETEEN'];
  const tens=['','','TWENTY','THIRTY','FORTY','FIFTY','SIXTY','SEVENTY','EIGHTY','NINETY'];
  const two=n=>n<20?ones[n]:tens[Math.floor(n/10)]+(n%10?' '+ones[n%10]:'');
  const three=n=>{const h=Math.floor(n/100),r=n%100;return (h?ones[h]+' HUNDRED':'')+(h&&r?' ':'')+(r?two(r):'');};
  function words(n){
    if(n===0)return 'ZERO';
    const b=Math.floor(n/1e9),m=Math.floor((n%1e9)/1e6),t=Math.floor((n%1e6)/1000),u=n%1000;
    let s='';
    if(b)s+=three(b)+' BILLION ';
    if(m)s+=three(m)+' MILLION ';
    if(t)s+=three(t)+' THOUSAND ';
    if(u)s+=three(u);
    return s.trim();
  }
  const intPart=Math.floor(num),cents=Math.round((num-intPart)*100);
  const c2=('0'+cents).slice(-2).split('').map(x=>ones[+x]).join(' ');
  return 'US DOLLARS '+words(intPart)+' AND CENTS '+c2+' ONLY';
}

function saveFreightRates(){
  const port=PORTS[$('freightPort').selectedIndex];
  if(!port) return;
  const rates={
    sea:Number($('freightSea').value),
    rail:Number($('freightRail').value),
    air:Number($('freightAir').value)
  };

  freightCalc();

}

function num(id,def){ const v=parseFloat($(id).value); return isNaN(v)?def:v; }
function calc(){
  const purchase=num('purchase',118.57), vat=num('vatRate',13)/100, reb=num('rebateRate',13)/100;
  const domestic=num('domestic',2.57), packCert=num('packCert',0.71), margin=num('margin',15)/100;
  const freight=freightCalc(), insRate=num('insRate',0.3)/100, addOn=1+num('insAddOn',10)/100;
  const qty=Math.max(1,Math.floor(num('qty',300))), samplePrice=num('samplePrice',280);
  const rebate=purchase/(1+vat)*reb;
  const actual=purchase-rebate;
  const totalCost=actual+domestic+packCert;
  const exw=(actual+packCert)*(1+margin);
  const fob=totalCost*(1+margin);
  const cif=(fob+freight)/(1-addOn*insRate);
  const ins=cif*addOn*insRate;
  const isSample=qty<50;
  const quoteExw=isSample?samplePrice-(fob-exw):exw;
  const quoteFob=isSample?samplePrice:fob;
  const quoteCif=(quoteFob+freight)/(1-addOn*insRate);
  const quoteIns=quoteCif*addOn*insRate;
  const quoteFreight=quoteFob+freight;
  const basis=INCOTERMS[TERM].basis;
  const unitPrice=basis==='exw'?quoteExw:basis==='fob'?quoteFob:basis==='freight'?quoteFreight:basis==='insured'?quoteCif:quoteFreight;
  return {
    purchase,rebate,actual,totalCost,domestic,packCert,profit:totalCost*margin,
    exw:quoteExw,fob:quoteFob,cif:quoteCif,ins:quoteIns,qty,freight,isSample,samplePrice,unitPrice
  };
}

function numToWords(num){
  const ones=['ZERO','ONE','TWO','THREE','FOUR','FIVE','SIX','SEVEN','EIGHT','NINE','TEN','ELEVEN','TWELVE','THIRTEEN','FOURTEEN','FIFTEEN','SIXTEEN','SEVENTEEN','EIGHTEEN','NINETEEN'];
  const tens=['','','TWENTY','THIRTY','FORTY','FIFTY','SIXTY','SEVENTY','EIGHTY','NINETY'];
  const two=n=>n<20?ones[n]:tens[Math.floor(n/10)]+(n%10?' '+ones[n%10]:'');
  const three=n=>{const h=Math.floor(n/100),r=n%100;return (h?ones[h]+' HUNDRED':'')+(h&&r?' ':'')+(r?two(r):'');};
  function words(n){
    if(n===0)return 'ZERO';
    const b=Math.floor(n/1e9),m=Math.floor((n%1e9)/1e6),t=Math.floor((n%1e6)/1000),u=n%1000;
    let s='';
    if(b)s+=three(b)+' BILLION ';
    if(m)s+=three(m)+' MILLION ';
    if(t)s+=three(t)+' THOUSAND ';
    if(u)s+=three(u);
    return s.trim();
  }
  const intPart=Math.floor(num),cents=Math.round((num-intPart)*100);
  const c2=('0'+cents).slice(-2).split('').map(x=>ones[+x]).join(' ');
  return 'US DOLLARS '+words(intPart)+' AND CENTS '+c2+' ONLY';
}

function update(){
  const r=calc();
  const usd=v=>v.toLocaleString('en-US',{minimumFractionDigits:2,maximumFractionDigits:2});
  $('rFOB').textContent=usd(r.fob);
  $('rCIF').textContent=usd(r.cif);
  $('rIns').textContent=['CIF','CIP'].includes(TERM)?usd(r.ins):'—';
  const iu=Math.round(r.unitPrice*100)/100;
  $('rTotal').textContent=usd(Math.round(iu*r.qty*100)/100);
  $('t_totalMeta').textContent=r.qty+' pcs · '+TERM+(r.isSample?(LANG==='en'?' · sample':' · 樣本'):'');
  const marginPct=num('margin',15);
  $('t_cProfit').textContent=(LANG==='en'?'Profit ':'利潤 ')+marginPct+'%';
  $('vPur').textContent=usd(r.purchase);
  $('vReb').textContent='−'+usd(r.rebate);
  $('vDom').textContent=usd(r.domestic);
  $('vPc').textContent=usd(r.packCert);
  $('vProfit').textContent=usd(r.profit);
  let html=DOC==='PI'?sheetHTML(r):contractHTML(r);
  if(TRANSPORT!=='sea'){
    const docEn=TRANSPORT==='air'?'air waybill (AWB)':'rail consignment note';
    const docZh=TRANSPORT==='air'?'空運提單（AWB）':'鐵路運單';
    html=html
      .replace(/vessel name, B\/L date/g,'flight/train details, transport document date')
      .replace(/船名、提單日期/g,'航班／列車資料、運輸單據日期')
      .replace(/full set clean on-board ocean B\/L|full set of clean on-board ocean B\/L|ocean bill of lading \(B\/L\)|ocean B\/L/g,docEn)
      .replace(/全套清潔已裝船海運提單|海運提單/g,docZh)
      .replace(/B\/L/g,docEn)
      .replace(/提單/g,docZh)
      .replace(/Sea freight/g,TRANSPORT==='air'?'Air freight':'Rail freight')
      .replace(/海運費/g,TRANSPORT==='air'?'空運費':'鐵路運費')
      .replace(/ocean carriage/g,'international carriage')
      .replace(/遠洋海運/g,'國際運輸')
      .replace(/裝船後/g,'發運後')
      .replace(/after loading of vessel/g,'after dispatch');
  }
  // 單據裡的縮寫不得被換行拆開
  $('sheet').innerHTML=html
    .replace(/L\/C/g,'<span class="nw">L/C</span>')
    .replace(/T\/T/g,'<span class="nw">T/T</span>')
    .replace(/B\/L/g,'<span class="nw">B/L</span>')
    .replace(/C\/O/g,'<span class="nw">C/O</span>');
}

function init(){
  const sel=$('port'), rateSel=$('freightPort');
  let lastR='';
  PORTS.forEach((p,i)=>{
    if(p.r!==lastR){
      const group=document.createElement('optgroup'); group.label=p.r; sel.appendChild(group);
      const rateGroup=document.createElement('optgroup'); rateGroup.label=p.r; rateSel.appendChild(rateGroup);
      lastR=p.r;
    }
    [sel.lastChild,rateSel.lastChild].forEach(group=>{
      const o=document.createElement('option'); o.textContent=p.n; o.dataset.en=p.en; o.dataset.idx=i; group.appendChild(o);
    });
  });
  const currencyMigration='qc_costs_usd_v1';
  if(!localStorage.getItem(currencyMigration)){
    const oldFx=parseFloat(localStorage.getItem('qc_fx')||'7');
    if(Number.isFinite(oldFx)&&oldFx>0){
      ['purchase','domestic','packCert'].forEach(k=>{
        const saved=localStorage.getItem('qc_'+k);
        const amount=Number(saved);
        if(saved!==null&&Number.isFinite(amount)){
          localStorage.setItem('qc_'+k,(amount/oldFx).toFixed(2));
        }
      });
    }
    localStorage.setItem(currencyMigration,'1');
  }
  // 樣本單基準單價 150 → 280（一次性遷移）
  if(!localStorage.getItem('qc_sampleprice_280_v1')){
    if(localStorage.getItem('qc_samplePrice')==='150') localStorage.setItem('qc_samplePrice','280');
    localStorage.setItem('qc_sampleprice_280_v1','1');
  }
  // 載入 localStorage
  const keys=['purchase','vatRate','rebateRate','domestic','packCert','margin','samplePrice','insRate','insAddOn','tradeTerm','sellerName','sellerAddr','sellerContact','sellerTel','sellerEmail','buyerName','buyerPO','buyerAddr','buyerContact','buyerTel','buyerEmail','beneficiary','bankName','swift','acct'];
  keys.forEach(k=>{ const v=localStorage.getItem('qc_'+k); if(v!==null) $(k).value=v; });
  if(Object.prototype.hasOwnProperty.call(INCOTERMS,$('tradeTerm').value)) TERM=$('tradeTerm').value;
  const savedTransport=localStorage.getItem('qc_transportMode');
  if(Object.prototype.hasOwnProperty.call(TRANSPORTS,savedTransport)) TRANSPORT=savedTransport;
  const savedRates=localStorage.getItem(FREIGHT_RATE_KEY);
  const storedRates=savedRates?JSON.parse(savedRates):{};
  PORTS.forEach(p=>{
    const sea=p.f/FREIGHT_DEFAULT_UNITS;
    const saved=storedRates[p.en];
    FREIGHT_RATES[p.en]=saved&&Number.isFinite(saved.sea)&&Number.isFinite(saved.rail)&&Number.isFinite(saved.air)
      &&saved.sea>=0&&saved.sea<=saved.rail&&saved.rail<=saved.air
      ? {sea:saved.sea,rail:saved.rail,air:saved.air}
      : {sea,rail:sea*1.8,air:sea*6};
  });
  if(!savedRates){
    const legacyRate=Number(localStorage.getItem('qc_containerRate'));
    const legacyUnits=Number(localStorage.getItem('qc_unitsPerContainer'));
    if(Number.isFinite(legacyRate)&&legacyRate>=0&&Number.isFinite(legacyUnits)&&legacyUnits>0){
      const sea=legacyRate/legacyUnits;
      FREIGHT_RATES[PORTS[0].en]={sea,rail:sea*1.8,air:sea*6};
    }
    localStorage.setItem(FREIGHT_RATE_KEY,JSON.stringify(FREIGHT_RATES));
  }
  $('transportMode').value=TRANSPORT;
  setTransportMode(TRANSPORT);
  const savedColor=localStorage.getItem('qc_productColor');
  if(savedColor&&Object.prototype.hasOwnProperty.call(PRODUCT_COLORS,savedColor)) PRODUCT_COLOR=savedColor;
  ['Black','White','Red'].forEach(c=>{
    const selected=c===PRODUCT_COLOR;
    $('color'+c).classList.toggle('on',selected);
    $('color'+c).setAttribute('aria-pressed',String(selected));
  });
  // 保存
  const save=()=>keys.forEach(k=>localStorage.setItem('qc_'+k,$(k).value));
  keys.forEach(id=>$(id).addEventListener('change',save));
  applyI18n();
  onPort();
}
window.addEventListener('DOMContentLoaded',init);
